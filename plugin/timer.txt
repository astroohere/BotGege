const version =
  "`timer v0.1`\nplugin ini berfungsi untuk menyetel timer dengan WhatsApp\n*catatan: jika bot crash/direstart maka timer yang sedang berjalan akan hilang*";

import {
  existsSync,
  mkdirSync,
  readdirSync,
  readFileSync,
  writeFileSync,
} from "fs";
import { join } from "path";

export default function setup(soket) {
  const timerDatabaseDir = "./database/timer";
  const timerDatabase = "timer.json";
  const timerDatabasePath = join(timerDatabaseDir, timerDatabase);

  if (!existsSync(timerDatabaseDir)) {
    mkdirSync(timerDatabaseDir);
  }
  if (!existsSync(timerDatabasePath)) {
    writeFileSync(timerDatabasePath);
  }

  setTimer(soket);
}
